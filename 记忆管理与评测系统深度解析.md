**关键属性** (代码位置 `models/memory.py:116-128`):

```python
@property
def retrieval_score(self) -> float:
    """综合检索得分 = 重要性 + 访问频率增益 + 时效性衰减"""
    score = self.importance_score
    # 访问频率增益: log10(access_count+1) * 0.1, 上限 0.3
    score += min(math.log10(self.access_count + 1) * 0.1, 0.3)
    # 时效性衰减: 线性衰减，168小时(7天)后归零，上限 0.2
    if self.last_accessed_at:
        hours = (datetime.now(timezone.utc) - self.last_accessed_at).total_seconds() / 3600
        score += max(0, 0.2 * (1 - hours / 168))
    return min(score, 1.0)
```

```python
@staticmethod
def generate_summary(content: str) -> str:
    MAX_SUMMARY_LENGTH = 200
    if len(content) <= 200:
        return content  # 不截断
    # 找句子分割符 "。 "
    sentences = content.split("。 ")
    if len(sentences) >= 2:
        # 首句 + 尾句(截断100字符)
        return sentences[0] + "。" + sentences[-1][:100]
    # 否则直接截断200字符 + "..."
    return content[:200] + "..."
```

#### 7.3.2 extract_keywords() — 关键词提取

```python
# core.py:122-150
@staticmethod
def extract_keywords(content: str, max_keywords: int = 5) -> list[str]:
    # 1. 正则提取所有 >=3 个字母的英文单词
    words = re.findall(r'\b[a-zA-Z]{3,}\b', content)
    # 2. 过滤停用词（~90个常见词 + "information", "error", "service"）
    filtered = [w for w in words if w.lower() not in STOPWORDS]
    # 3. 过滤纯数字
    filtered = [w for w in filtered if not w.isdigit()]
    # 4. 按频率排序，取 top max_keywords
    freq = Counter(filtered)
    return [word for word, _ in freq.most_common(max_keywords)]
```

```python
async def _find_similar(self, entry: MemoryEntry) -> MemoryEntry | None:
    # 搜索与 entry.content 最相似的一条已有记忆
    results = await self._storage.search(
        query=entry.content,
        top_k=1,
        filters={
            "memory_type": entry.memory_type.value,  # "episodic"
            "source_agent": entry.source_agent,      # "rca_agent"
        },
    )

    if results:
        stored_vec = results[0].content_vector
        entry_vec = entry.content_vector
        similarity = self._cosine_similarity(stored_vec, entry_vec)
        # 余弦相似度 = dot(a, b) / (sqrt(sum(a²)) * sqrt(sum(b²)))

        if similarity >= self._similarity_threshold:  # 默认 0.85
            return results[0]  # 找到相似记忆！

    return None
```

```python
# long_term.py:79-110
def keyword_search_score(query: str, entry: MemoryEntry) -> float:
    score = 0.0
    query_lower = query.lower()
    content_lower = entry.content.lower()

    # 1. 精确匹配: +0.8
    if query_lower in content_lower:
        score += 0.8  # 不完全匹配，跳过

    # 2. 词频统计
    query_words = query_lower.split()
    content_words = content_lower.split()
    # query: ["order-service", "cpu", "使用率", "突然升高", ...]
    # content: ["[rcAagent]", "root", "cause:", "近期部署...", ...]
    for word in query_words:
        tf = content_words.count(word) / max(len(content_words), 1)
        score += tf * 2.0
    # 假设 "order-service" 出现 1 次, "cpu" 出现 0 次:
    # → score += (1/15)*2.0 + 0 = 0.133

    # 3. 标签匹配: +0.5 per match
    for tag in entry.tags:
        if tag in query_lower:
            score += 0.5
    # "cpu" 匹配, "order-service" 匹配 → +1.0

    # 4. 摘要匹配: +0.3
    if any(word in entry.summary.lower() for word in query_words):
        score += 0.3

    return min(score, 2.0)  # 上限 2.0
    # 实际 ≈ 0.8 + 0.133 + 1.0 + 0.3 = 2.233 → 钳制为 2.0
```

```python
def time_decay_score(entry, current_time=None, half_life_days=30.0):
    # 使用 last_accessed_at 或 created_at (取较新的)
    reference_time = entry.last_accessed_at or entry.created_at
    # → 2026-06-29T10:01:23Z (刚创建)

    age_days = max(0, (current_time - reference_time).total_seconds() / 86400)
    # → 刚创建，age_days ≈ 0

    # 衰减因子: 0.5 ^ (age_days / half_life_days)
    decay_factor = 0.5 ** (0 / 30) = 1.0

    # 访问增益: min(access_count * 0.05, 0.3)
    access_boost = min(0 * 0.05, 0.3) = 0.0

    # 最终得分
    score = importance_score * decay_factor + access_boost
    # = 0.85 * 1.0 + 0.0 = 0.85

    return score
```

```python
# long_term.py:520-580
async def merge_similar(self, similarity_threshold: float | None = None) -> int:
    threshold = similarity_threshold or self._similarity_threshold  # 默认 0.85

    all_entries = await self._storage.get_by_type(
        memory_level=MemoryLevel.LONG_TERM, limit=5000
    )

    # 按 memory_type 分组
    groups: dict[str, list[MemoryEntry]] = {}
    for entry in all_entries:
        groups.setdefault(entry.memory_type.value, []).append(entry)

    merged = 0
    for group_entries in groups.values():
        for i, entry_a in enumerate(group_entries):
            for entry_b in group_entries[i+1:]:
                if entry_a.content_vector and entry_b.content_vector:
                    sim = self._cosine_similarity(
                        entry_a.content_vector, entry_b.content_vector
                    )
                    if sim >= threshold:
                        # 合并: a 吸收 b 的内容
                        entry_a.content = self._merge_content(entry_a, entry_b)
                        entry_a.importance_score = max(
                            entry_a.importance_score, entry_b.importance_score
                        )
                        entry_a.tags = list(set(entry_a.tags) | set(entry_b.tags))
                        entry_a.access_count += entry_b.access_count
                        await self._storage.delete(entry_b.memory_id)
                        merged += 1

    return merged
```
